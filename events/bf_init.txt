# BrainFuck Machine

# Machine initialization

namespace = bfvm

# Setup vm reference
country_event = {
    id = bfvm.1

    hide_window = yes
    is_triggered_only = yes

    immediate = {
        save_global_event_target_as = bf_vm_root

        capital_scope = {
            planet = {
                save_global_event_target_as = bf_vm

                create_node = yes

                last_created_army = {
                    save_global_event_target_as = bf_vm_null
                }

                planet_event = { id = bfvm.2 }
            }
        }
    }
}

# Setup memory
planet_event = {
    id = bfvm.2

    hide_window = yes
    is_triggered_only = yes

    immediate = {
        # First we need to create the memory
        create_node = yes
        last_created_army = {
            save_global_event_target_as = bf_vm_memory_head
            save_global_event_target_as = bf_vm_memory_construct_cursor
        }

        # Then we initialize 31 additional nodes, for 32 cells total
        while = {
            count = 31

            linked_list_extend_after = {
                NODE = bf_vm_memory_construct_cursor
                NULL = bf_vm_null
            }
        }

        event_target:bf_vm_memory_construct_cursor = {
            save_global_event_target_as = bf_vm_memory_tail
        }

        event_target:bf_vm_memory_head = {
            save_global_event_target_as = bf_vm_memory_cursor
        }
    }
}
