# Add one to the cursor's pointing node
planet_event = {
    id = bfvm.11

    hide_window = yes
    is_triggered_only = yes

    immediate = {
        event_target:bf_vm_memory_cursor = {
            add_variable = {
                which = bf_ll_value
                value = 1
            }

            if = {
                limit = {
                    check_variable = {
                        which = bf_ll_value
                        value > 255
                    }
                }

                subtract_variable = {
                    which = bf_ll_value
                    value = 256
                }
            }
        }
    }
}

# Subtract one to the cursor's pointing node
planet_event = {

                        which = bf_ll_value
                        value < 0
                    }
                }

                add_variable = {
                    which = bf_ll_value
                    value = 256
                }
            }
        }
    }
}

# Move the memory cursor right, extending the tape at its tail
planet_event = {
    id = bfvm.14

    hide_window = yes
    is_triggered_only = yes

    immediate = {
        event_target:bf_vm_memory_cursor = {
            if = {
                limit = {
                    is_same_value = event_target:bf_vm_memory_tail
                }

                linked_list_extend_after = {
                    NODE = bf_vm_memory_cursor
                    NULL = bf_vm_null
                }

                event_target:bf_vm_memory_cursor = {
                    save_global_event_target_as = bf_vm_memory_tail
                }
            }
            else = {
                event_target:bf_ll_next@this = {
                    save_global_event_target_as = bf_vm_memory_cursor
                }
            }
        }
    }
}

# Move the memory cursor left, extending the tape at its head

    id = bfvm.15

    hide_window = yes
    is_triggered_only = yes
