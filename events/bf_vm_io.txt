# Add one to the cursor's pointing node
planet_event = {
    id = bfvm.11

    hide_window = yes
    is_triggered_only = yes

    immediate = {
        event_target:bf_vm_memory_cursor = {
            add_variable = {
                which = bf_ll_value
                value = 1
            }

            if {
                limit = {
                    check_variable = {
                        which = bf_ll_value
                        value > 255
                    }
                }

                subtract_variable = {
                    which = bf_ll_value
                    value = 256
                }
            }
        }
    }
}

# Subtract one to the cursor's pointing node
planet_event = {
    id = bfvm.12

    hide_window = yes
    is_triggered_only = yes

    immediate = {
        event_target:bf_vm_memory_cursor = {
            subtract_variable = {
                which = bf_ll_value
                value = 1
            }

            if {
                limit = {
                    check_variable = {
                        which = bf_ll_value
                        value < 0
                    }
                }

                add_variable = {
                    which = bf_ll_value
                    value = 256
                }
            }
        }
    }
}

# Set the node to zero
planet_event = {

    is_triggered_only = yes

    immediate = {
        event_target:bf_vm_memory_cursor = {
            set_variable = {
                which = bf_ll_value
                value = 0
            }
        }
    }
}
